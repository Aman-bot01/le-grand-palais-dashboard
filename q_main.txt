DEFINE
TABLE 'Staffing Lever' = SELECTCOLUMNS ( GENERATESERIES ( 0, 30, 1 ), "Staffing Lever", [Value] )   // % of delayed calls fixed in Morning/Afternoon
TABLE 'Area Lever' = SELECTCOLUMNS ( GENERATESERIES ( 0, 30, 1 ), "Area Lever", [Value] )       // % improvement in high-delay municipalities
TABLE 'Station Lever' = SELECTCOLUMNS ( GENERATESERIES ( 0, 30, 1 ), "Station Lever", [Value] )    // % improvement at slower stations
TABLE 'Ramp Months' = SELECTCOLUMNS ( GENERATESERIES ( 1, 12, 1 ), "Ramp Months", [Value] )      // months until changes reach full effect
TABLE 'Trend Window' = SELECTCOLUMNS ( GENERATESERIES ( 6, 24, 1 ), "Trend Window", [Value] )     // months of history used to fit the baseline trend
TABLE 'Time Impact' = SELECTCOLUMNS ( GENERATESERIES ( 0, 100, 5 ), "Time Impact", [Value] )     // % of the delay reduction that also shortens response time

Milestone =
DATATABLE (
    "Months Ahead", INTEGER,
    "Milestone", STRING,
    { { 3, "3 months" }, { 6, "6 months" }, { 9, "9 months" }, { 12, "12 months" } }
)
// After creating: Add each slider table's column as a Slicer (single select, slider style).
// Defaults used when nothing is selected: Staffing 10, Area 8, Station 8, Ramp 6, Window 12, Time impact 50.


//
MEASURE 'Forecast Month'[Trend Window Months] = SELECTEDVALUE ( 'Trend Window'[Trend Window], 12 )
MEASURE 'Forecast Month'[Staffing Lever Pct] = SELECTEDVALUE ( 'Staffing Lever'[Staffing Lever], 10 ) / 100
MEASURE 'Forecast Month'[Area Lever Pct] = SELECTEDVALUE ( 'Area Lever'[Area Lever], 8 ) / 100
MEASURE 'Forecast Month'[Station Lever Pct] = SELECTEDVALUE ( 'Station Lever'[Station Lever], 8 ) / 100
MEASURE 'Forecast Month'[Ramp Months Value] = SELECTEDVALUE ( 'Ramp Months'[Ramp Months], 6 )
MEASURE 'Forecast Month'[Time Impact Pct] = SELECTEDVALUE ( 'Time Impact'[Time Impact], 50 ) / 100

Hist Calls =
VAR y = MAX ( 'Forecast Month'[Year] )
VAR m = MAX ( 'Forecast Month'[Month Num] )
RETURN CALCULATE ( COUNTROWS ( Call_Master ), Call_Master[Year] = y, Call_Master[Month Number] = m )

Hist Eligible =
VAR y = MAX ( 'Forecast Month'[Year] )
VAR m = MAX ( 'Forecast Month'[Month Num] )
RETURN
    CALCULATE (
        COUNTROWS ( Call_Master ),
        NOT ( ISBLANK ( Call_Master[Delay Threshold] ) ),
        Call_Master[Year] = y, Call_Master[Month Number] = m
    )

Hist Delay Rate =
// Delay rate on calls that have a benchmark (Patient Category assigned) — avoids the blank-category distortion
VAR y = MAX ( 'Forecast Month'[Year] )
VAR m = MAX ( 'Forecast Month'[Month Num] )
RETURN
    DIVIDE (
        CALCULATE ( COUNTROWS ( Call_Master ), Call_Master[Delay Flag] = 1, Call_Master[Year] = y, Call_Master[Month Number] = m ),
        [Hist Eligible]
    )

Hist Median Response =
VAR y = MAX ( 'Forecast Month'[Year] )
VAR m = MAX ( 'Forecast Month'[Month Num] )
RETURN CALCULATE ( MEDIAN ( Call_Master[Response time minutes] ), Call_Master[Year] = y, Call_Master[Month Number] = m )

Last Hist Index =
CALCULATE ( MAX ( 'Forecast Month'[Month Index] ), REMOVEFILTERS ( 'Forecast Month' ), 'Forecast Month'[Is Forecast] = FALSE () )
MEASURE 'Forecast Month'[Months Ahead Value] = MAX ( 'Forecast Month'[Months Ahead] )

Baseline Delay Rate =
VAR LastIdx = [Last Hist Index]
VAR Hist =
    FILTER ( ALL ( 'Forecast Month' ),
        'Forecast Month'[Month Index] > LastIdx - [Trend Window Months] && 'Forecast Month'[Month Index] <= LastIdx )
VAR T   = ADDCOLUMNS ( Hist, "@y", [Hist Delay Rate] )
VAR Fit = LINESTX ( T, [@y], 'Forecast Month'[Month Index] )
RETURN
    MIN ( 0.95, MAX ( 0, MAXX ( Fit, [Intercept] ) + MAXX ( Fit, [Slope1] ) * MAX ( 'Forecast Month'[Month Index] ) ) )

Baseline Calls =
VAR LastIdx = [Last Hist Index]
VAR Hist =
    FILTER ( ALL ( 'Forecast Month' ),
        'Forecast Month'[Month Index] > LastIdx - [Trend Window Months] && 'Forecast Month'[Month Index] <= LastIdx )
VAR T   = ADDCOLUMNS ( Hist, "@y", [Hist Calls] )
VAR Fit = LINESTX ( T, [@y], 'Forecast Month'[Month Index] )
RETURN
    MAX ( 0, MAXX ( Fit, [Intercept] ) + MAXX ( Fit, [Slope1] ) * MAX ( 'Forecast Month'[Month Index] ) )

Baseline Median Response =
VAR LastIdx = [Last Hist Index]
VAR Hist =
    FILTER ( ALL ( 'Forecast Month' ),
        'Forecast Month'[Month Index] > LastIdx - [Trend Window Months] && 'Forecast Month'[Month Index] <= LastIdx )
VAR T   = ADDCOLUMNS ( Hist, "@y", [Hist Median Response] )
VAR Fit = LINESTX ( T, [@y], 'Forecast Month'[Month Index] )
RETURN
    MAX ( 0, MAXX ( Fit, [Intercept] ) + MAXX ( Fit, [Slope1] ) * MAX ( 'Forecast Month'[Month Index] ) )

Delayed L12M =
VAR MaxD   = CALCULATE ( MAX ( Call_Master[T1_CallReceived] ), ALL ( Call_Master ) )
VAR StartD = EDATE ( DATE ( YEAR ( MaxD ), MONTH ( MaxD ), 1 ), -11 )
RETURN CALCULATE ( COUNTROWS ( Call_Master ), Call_Master[Delay Flag] = 1, Call_Master[T1_CallReceived] >= StartD )

Share Peak Hours =
// Recommendation 1: review afternoon/morning staffing
VAR MaxD   = CALCULATE ( MAX ( Call_Master[T1_CallReceived] ), ALL ( Call_Master ) )
VAR StartD = EDATE ( DATE ( YEAR ( MaxD ), MONTH ( MaxD ), 1 ), -11 )
RETURN
    DIVIDE (
        CALCULATE ( COUNTROWS ( Call_Master ), Call_Master[Delay Flag] = 1, Call_Master[T1_CallReceived] >= StartD,
            Call_Master[Hour Group] IN { "Morning", "Afternoon" } ),
        [Delayed L12M]
    )

Share High Delay Areas =
// Recommendation 2: reassess service in high-delay areas
VAR MaxD   = CALCULATE ( MAX ( Call_Master[T1_CallReceived] ), ALL ( Call_Master ) )
VAR StartD = EDATE ( DATE ( YEAR ( MaxD ), MONTH ( MaxD ), 1 ), -11 )
RETURN
    DIVIDE (
        CALCULATE ( COUNTROWS ( Call_Master ), Call_Master[Delay Flag] = 1, Call_Master[T1_CallReceived] >= StartD,
            Call_Master[PickupLowerTierDescription] IN { "WhiteWater", "Greater Madawaska", "North Algona Wilberforce", "McNab-Braeside" } ),
        [Delayed L12M]
    )

Share Slow Stations =
// Recommendation 4: review slower-response stations
VAR MaxD   = CALCULATE ( MAX ( Call_Master[T1_CallReceived] ), ALL ( Call_Master ) )
VAR StartD = EDATE ( DATE ( YEAR ( MaxD ), MONTH ( MaxD ), 1 ), -11 )
RETURN
    DIVIDE (
        CALCULATE ( COUNTROWS ( Call_Master ), Call_Master[Delay Flag] = 1, Call_Master[T1_CallReceived] >= StartD,
            Call_Master[RosteredFromStationName] IN { "BARRY'S BAY", "RENFREW", "EGANVILLE", "CHALK RIVER" } ),
        [Delayed L12M]
    )

// Levers are combined multiplicatively so overlapping scopes are not double counted.
Combined Effect =
1 - ( 1 - [Share Peak Hours] * [Staffing Lever Pct] )
      * ( 1 - [Share High Delay Areas] * [Area Lever Pct] )
      * ( 1 - [Share Slow Stations] * [Station Lever Pct] )
MEASURE 'Forecast Month'[Ramp Factor] = MIN ( 1, DIVIDE ( [Months Ahead Value], [Ramp Months Value], 0 ) )
MEASURE 'Forecast Month'[Scenario Delay Rate] = [Baseline Delay Rate] * ( 1 - [Combined Effect] * [Ramp Factor] )

Scenario Median Response =
[Baseline Median Response] * ( 1 - [Combined Effect] * [Time Impact Pct] * [Ramp Factor] )

Eligible Share L12M =
VAR MaxD   = CALCULATE ( MAX ( Call_Master[T1_CallReceived] ), ALL ( Call_Master ) )
VAR StartD = EDATE ( DATE ( YEAR ( MaxD ), MONTH ( MaxD ), 1 ), -11 )
RETURN
    DIVIDE (
        CALCULATE ( COUNTROWS ( Call_Master ), NOT ( ISBLANK ( Call_Master[Delay Threshold] ) ), Call_Master[T1_CallReceived] >= StartD ),
        CALCULATE ( COUNTROWS ( Call_Master ), Call_Master[T1_CallReceived] >= StartD )
    )
MEASURE 'Forecast Month'[Baseline Delayed Calls] = [Baseline Calls] * [Eligible Share L12M] * [Baseline Delay Rate]
MEASURE 'Forecast Month'[Scenario Delayed Calls] = [Baseline Calls] * [Eligible Share L12M] * [Scenario Delay Rate]
MEASURE 'Forecast Month'[Delayed Calls Avoided] = [Baseline Delayed Calls] - [Scenario Delayed Calls]

Actual Delay Rate =
IF ( NOT ( MAX ( 'Forecast Month'[Is Forecast] ) ), [Hist Delay Rate] )

Baseline Line =
VAR Idx = MAX ( 'Forecast Month'[Month Index] )
VAR L   = [Last Hist Index]
RETURN SWITCH ( TRUE (), Idx < L, BLANK (), Idx = L, [Hist Delay Rate], [Baseline Delay Rate] )

Scenario Line =
VAR Idx = MAX ( 'Forecast Month'[Month Index] )
VAR L   = [Last Hist Index]
RETURN SWITCH ( TRUE (), Idx < L, BLANK (), Idx = L, [Hist Delay Rate], [Scenario Delay Rate] )

Actual Median Response =
IF ( NOT ( MAX ( 'Forecast Month'[Is Forecast] ) ), [Hist Median Response] )

Baseline Response Line =
VAR Idx = MAX ( 'Forecast Month'[Month Index] )
VAR L   = [Last Hist Index]
RETURN SWITCH ( TRUE (), Idx < L, BLANK (), Idx = L, [Hist Median Response], [Baseline Median Response] )

Scenario Response Line =
VAR Idx = MAX ( 'Forecast Month'[Month Index] )
VAR L   = [Last Hist Index]
RETURN SWITCH ( TRUE (), Idx < L, BLANK (), Idx = L, [Hist Median Response], [Scenario Median Response] )

MS Baseline Delay Rate =
CALCULATE ( [Baseline Delay Rate], 'Forecast Month'[Months Ahead] = SELECTEDVALUE ( Milestone[Months Ahead] ) )

MS Scenario Delay Rate =
CALCULATE ( [Scenario Delay Rate], 'Forecast Month'[Months Ahead] = SELECTEDVALUE ( Milestone[Months Ahead] ) )

MS Benchmark Met Scenario =
1 - [MS Scenario Delay Rate]

MS Median Response Baseline =
CALCULATE ( [Baseline Median Response], 'Forecast Month'[Months Ahead] = SELECTEDVALUE ( Milestone[Months Ahead] ) )

MS Median Response Scenario =
CALCULATE ( [Scenario Median Response], 'Forecast Month'[Months Ahead] = SELECTEDVALUE ( Milestone[Months Ahead] ) )

MS Delayed Calls Avoided (cumulative) =
VAR M = SELECTEDVALUE ( Milestone[Months Ahead] )
RETURN
    SUMX (
        FILTER ( ALL ( 'Forecast Month' ), 'Forecast Month'[Months Ahead] >= 1 && 'Forecast Month'[Months Ahead] <= M ),
        [Delayed Calls Avoided]
    )

MS Delay Rate Improvement (pts) =
[MS Baseline Delay Rate] - [MS Scenario Delay Rate]
MEASURE 'Forecast Month'[Effect Staffing] = [Share Peak Hours] * [Staffing Lever Pct]
MEASURE 'Forecast Month'[Effect Areas] = [Share High Delay Areas] * [Area Lever Pct]
MEASURE 'Forecast Month'[Effect Stations] = [Share Slow Stations] * [Station Lever Pct]

Forecast Headline =
"In 12 months, the delay rate moves from " & FORMAT ( CALCULATE ( [Hist Delay Rate], 'Forecast Month'[Month Index] = [Last Hist Index] ), "0.0%" )
    & " (last actual month) to " & FORMAT ( CALCULATE ( [Baseline Delay Rate], 'Forecast Month'[Months Ahead] = 12 ), "0.0%" )
    & " if nothing changes, or " & FORMAT ( CALCULATE ( [Scenario Delay Rate], 'Forecast Month'[Months Ahead] = 12 ), "0.0%" )
    & " with the recommendations applied."
EVALUATE
SELECTCOLUMNS(
 FILTER('Forecast Month', 'Forecast Month'[Month Index] >= 33),
 "Label",'Forecast Month'[Label], "Idx",'Forecast Month'[Month Index],
 "Actual",[Actual Delay Rate],"Base",[Baseline Line],"Scen",[Scenario Line],
 "BaseResp",[Baseline Response Line],"ScenResp",[Scenario Response Line],
 "BaseCalls",[Baseline Calls],"Avoided",[Delayed Calls Avoided])
ORDER BY [Idx]
